(function() { const deck = document.getElementById('deck'); const slides = Array.from(document.querySelectorAll('.slide')); const total = slides.length; const progress = document.getElementById('progress'); const counterNow = document.getElementById('counterNow'); const counterAll = document.getElementById('counterAll'); const prevBtn = document.getElementById('prevBtn'); const nextBtn = document.getElementById('nextBtn'); const hint = document.getElementById('hint'); if (!deck || total === 0) return; counterAll.textContent = String(total).padStart(2, '0'); slides.forEach((_, i) => { const dot = document.createElement('button'); dot.className = 'dot'; dot.setAttribute('aria-label', 'Przejdź do slajdu ' + (i + 1)); dot.addEventListener('click', () => goTo(i)); progress.appendChild(dot); }); const dots = Array.from(progress.querySelectorAll('.dot')); let index = 0; let animating = false; function render() { deck.style.transform = 'translateX(-' + (index * 100) + 'vw)'; slides.forEach((s, i) => s.classList.toggle('is-active', i === index)); dots.forEach((d, i) => d.classList.toggle('active', i === index)); counterNow.textContent = String(index + 1).padStart(2, '0'); prevBtn.disabled = index === 0; nextBtn.disabled = index === total - 1; } function goTo(i) { if (animating) return; i = Math.max(0, Math.min(total - 1, i)); if (i === index) return; index = i; animating = true; render(); setTimeout(() => { animating = false; }, 760); dismissHint(); } function next() { goTo(index + 1); } function prev() { goTo(index - 1); } function dismissHint() { if (hint) hint.classList.add('hidden'); } prevBtn.addEventListener('click', prev); nextBtn.addEventListener('click', next); window.addEventListener('keydown', (e) => { if (e.key === 'ArrowRight' || e.key === ' ' || e.key === 'PageDown') { e.preventDefault(); next(); } else if (e.key === 'ArrowLeft' || e.key === 'PageUp') { e.preventDefault(); prev(); } else if (e.key === 'Home') { e.preventDefault(); goTo(0); } else if (e.key === 'End') { e.preventDefault(); goTo(total - 1); } }); let wheelLock = false; window.addEventListener('wheel', (e) => { const activeSlide = slides[index]; const isScrollable = activeSlide.scrollHeight > activeSlide.clientHeight; // Zezwalaj na naturalny scroll wewnątrz slajdu, jeśli treść się nie mieści if (isScrollable) { const atTop = activeSlide.scrollTop === 0; const atBottom = Math.abs(activeSlide.scrollHeight - activeSlide.clientHeight - activeSlide.scrollTop) < 2; if ((e.deltaY < 0 && !atTop) || (e.deltaY > 0 && !atBottom)) { return; } } e.preventDefault(); if (wheelLock || animating) return; const delta = Math.abs(e.deltaX) > Math.abs(e.deltaY) ? e.deltaX : e.deltaY; if (Math.abs(delta) < 22) return; wheelLock = true; if (delta > 0) next(); else prev(); setTimeout(() => { wheelLock = false; }, 500); }, { passive: false }); let touchStartX = null; let touchStartY = null; window.addEventListener('touchstart', (e) => { touchStartX = e.touches[0].clientX; touchStartY = e.touches[0].clientY; }, { passive: true }); window.addEventListener('touchend', (e) => { if (touchStartX === null || touchStartY === null) return; const diffX = touchStartX - e.changedTouches[0].clientX; const diffY = touchStartY - e.changedTouches[0].clientY; // Przełączaj slajd tylko przy wyraźnym geście poziomym (większym niż pionowy) if (Math.abs(diffX) > Math.abs(diffY) && Math.abs(diffX) > 40) { if (diffX > 0) next(); else prev(); } touchStartX = null; touchStartY = null; }, { passive: true }); render(); })();
